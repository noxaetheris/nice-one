function v1:CreateWindow(config)
	config = config or {}
	local winTitle = tostring(config.Name or "specter")
	local win = Library:CreateWindow(winTitle)

	local v2 = {}
	v2.Window = win

	function v2:SelectTab(tabObj)
		if tabObj and tabObj._activate then
			pcall(tabObj._activate)
		end
	end

	local currentToggleKey = Enum.KeyCode.RightShift

	function v2:SetToggleKey(keyName)
		pcall(function()
			local kc = typeof(keyName) == "EnumItem" and keyName or Enum.KeyCode[tostring(keyName)]
			if kc then
				currentToggleKey = kc
				if win then
					win.ToggleKey = kc
				end
			end
		end)
	end

	if config and config.ToggleUIKeybind then
		v2:SetToggleKey(config.ToggleUIKeybind)
	else
		v2:SetToggleKey("RightShift")
	end

	function v2:Toggle()
		pcall(function()
			if win and win.Toggle then
				win:Toggle()
			elseif self.Window and self.Window.Toggle then
				self.Window:Toggle()
			end
		end)
	end

	pcall(function()
		local uis = game:GetService("UserInputService")
		uis.InputBegan:Connect(function(input, gpe)
			if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
			if uis:GetFocusedTextBox() then return end
			if input.KeyCode == currentToggleKey then
				v2:Toggle()
			end
		end)
	end)

	pcall(function()
		local function sanitizeText(desc)
			if desc:IsA("TextLabel") or desc:IsA("TextButton") or desc:IsA("TextBox") then
				desc.AutoLocalize = false
				local function applyReplacement()
					local txt = desc.Text
					if txt == "Impostazioni" then
						desc.Text = "Settings"
					elseif txt == "cezar" then
						desc.Text = "Black"
					elseif txt == "Graphite" then
						desc.Text = "Dark Gray"
					elseif txt == "Contrast" then
						desc.Text = "Silver"
					elseif txt == "Paper" then
						desc.Text = "White"
					elseif txt:find("[Pp]unishment%s*[Hh]ub") then
						desc.Text = txt:gsub("[Pp]unishment%s*[Hh]ub", "specter")
					elseif txt:find("[Pp]unishment") then
						desc.Text = txt:gsub("[Pp]unishment", "specter")
					end
				end
				applyReplacement()
				desc:GetPropertyChangedSignal("Text"):Connect(applyReplacement)
			end
		end

		local function handleGui(gui)
			if gui and (gui.Name == "cezarUI" or gui:IsA("ScreenGui")) then
				gui.AutoLocalize = false
				for _, desc in ipairs(gui:GetDescendants()) do
					sanitizeText(desc)
				end
				gui.DescendantAdded:Connect(sanitizeText)
			end
		end

		local pGui = LocalPlayer:FindFirstChildOfClass("PlayerGui") or LocalPlayer:WaitForChild("PlayerGui")
		for _, g in ipairs(pGui:GetChildren()) do
			handleGui(g)
		end
		pGui.ChildAdded:Connect(handleGui)

		pcall(function()
			local cGui = (gethui and gethui()) or game:GetService("CoreGui")
			if cGui then
				for _, g in ipairs(cGui:GetChildren()) do
					handleGui(g)
				end
				cGui.ChildAdded:Connect(handleGui)
			end
		end)
	end)

	function v2:CreateTab(tConfig)
		local tName = "Tab"
		local icon = nil
		if type(tConfig) == "table" then
			tName = tostring(tConfig.Name or tConfig.Title or tConfig[1] or "Tab")
			icon = tConfig.Icon
		else
			tName = tostring(tConfig or "Tab")
		end

		local pTab = win:CreateTab(tName, icon)
		local tabObj = {}
		tabObj.RawTab = pTab

		local tData = (win.Tabs and win.Tabs[#win.Tabs])
		if tData and tData.TitleLabel then
			tData.TitleLabel.AutoLocalize = false
			tData.TitleLabel.Text = tName
			if tName == "Settings" then
				tData.TitleLabel:GetPropertyChangedSignal("Text"):Connect(function()
					if tData.TitleLabel.Text == "Impostazioni" then
						tData.TitleLabel.AutoLocalize = false
						tData.TitleLabel.Text = "Settings"
					end
				end)
			end

			tabObj._activate = function()
				pcall(function()
					for _, v in pairs(win.Tabs) do
						if v ~= tData and v.Page then v.Page.Visible = false end
					end
					win.CurrentTab = tData
					if tData.Page then tData.Page.Visible = true end
				end)
			end
		end

		function tabObj:CreateSection(sTitle)
			local title = ""
			if type(sTitle) == "table" then
				title = tostring(sTitle.Title or sTitle.Name or sTitle[1] or "")
			else
				title = tostring(sTitle or "")
			end
			if pTab and pTab.CreateSection then
				return pTab:CreateSection(title)
			end
		end

		function tabObj:CreateParagraph(pConfig)
			local pTitle = ""
			local pContent = ""
			if type(pConfig) == "table" then
				pTitle = tostring(pConfig.Title or pConfig.Name or "")
				pContent = tostring(pConfig.Content or pConfig.Text or pConfig.Desc or "")
			else
				pTitle = tostring(pConfig or "")
			end
			if pTab and pTab.CreateParagraph then
				return pTab:CreateParagraph(pTitle, pContent)
			end
		end

		function tabObj:CreateLabel(lConfig)
			local text = ""
			local updateFunc = nil
			local updateRate = nil
			if type(lConfig) == "table" then
				text = tostring(lConfig.Text or lConfig.Title or lConfig.Name or "")
				updateFunc = lConfig.Update
				updateRate = tonumber(lConfig.UpdateRate) or 1
			else
				text = tostring(lConfig or "")
			end
			local lbl = nil
			if pTab and pTab.CreateLabel then
				lbl = pTab:CreateLabel(text)
			end
			if type(updateFunc) == "function" and lbl then
				task.spawn(function()
					while true do
						task.wait(updateRate or 1)
						local ok, newTxt = pcall(updateFunc)
						if ok and newTxt then
							pcall(function()
								if lbl.SetText then
									lbl:SetText(tostring(newTxt))
								elseif lbl.Text then
									lbl.Text = tostring(newTxt)
								end
							end)
						end
					end
				end)
			end
			return lbl or { Set = function() end }
		end

		function tabObj:CreateButton(bConfig)
			local bName = "Button"
			local bDesc = ""
			local bCallback = function() end
			if type(bConfig) == "table" then
				bName = tostring(bConfig.Name or bConfig.Title or "Button")
				bDesc = tostring(bConfig.Desc or bConfig.Description or "")
				bCallback = bConfig.Callback or function() end
			else
				bName = tostring(bConfig or "Button")
			end
			if pTab and pTab.CreateButton then
				return pTab:CreateButton(bName, bDesc, nil, bCallback)
			end
			return { Set = function() end }
		end

		function tabObj:CreateToggle(tData)
			local tName = "Toggle"
			local tDesc = ""
			local tVal = false
			local tCallback = function() end
			if type(tData) == "table" then
				tName = tostring(tData.Name or tData.Title or "Toggle")
				tDesc = tostring(tData.Desc or tData.Description or "")
				tVal = tData.CurrentValue == true or tData.Default == true
				tCallback = tData.Callback or function() end
			else
				tName = tostring(tData or "Toggle")
			end

			local current = tVal
			if pTab and pTab.CreateToggle then
				pTab:CreateToggle(tName, tDesc, tVal, function(newVal)
					current = newVal
					pcall(tCallback, newVal)
				end)
			end

			local ctrl = {
				CurrentValue = current,
				Set = function(self, val)
					current = val == true
					self.CurrentValue = current
					pcall(tCallback, current)
				end
			}
			return ctrl
		end

		function tabObj:CreateSlider(sData)
			local sName = "Slider"
			local minVal = 0
			local maxVal = 100
			local defaultVal = 50
			local sCallback = function() end
			if type(sData) == "table" then
				sName = tostring(sData.Name or sData.Title or "Slider")
				local range = sData.Range or {0, 100}
				minVal = tonumber(range[1]) or 0
				maxVal = tonumber(range[2]) or 100
				defaultVal = tonumber(sData.CurrentValue or sData.Default) or minVal
				sCallback = sData.Callback or function() end
			else
				sName = tostring(sData or "Slider")
			end

			local current = defaultVal
			if pTab and pTab.CreateSlider then
				pTab:CreateSlider(sName, minVal, maxVal, defaultVal, function(newVal)
					current = newVal
					pcall(sCallback, newVal)
				end)
			end

			local ctrl = {
				CurrentValue = current,
				Set = function(self, val)
					local num = tonumber(val) or current
					current = math.clamp(num, minVal, maxVal)
					self.CurrentValue = current
					pcall(sCallback, current)
				end
			}
			return ctrl
		end

		function tabObj:CreateDropdown(dData)
			local dName = "Dropdown"
			local dDesc = ""
			local options = {}
			local defaultOpt = nil
			local dCallback = function() end
			if type(dData) == "table" then
				dName = tostring(dData.Name or dData.Title or "Dropdown")
				dDesc = tostring(dData.Desc or dData.Description or "")
				options = dData.Options or {}
				defaultOpt = dData.CurrentOption or dData.Default or (options[1] or "")
				dCallback = dData.Callback or function() end
			else
				dName = tostring(dData or "Dropdown")
			end

			local current = defaultOpt
			if pTab and pTab.CreateDropdown then
				pTab:CreateDropdown(dName, dDesc, options, function(selected)
					current = selected
					pcall(dCallback, selected)
				end)
			end

			local ctrl = {
				CurrentOption = current,
				Set = function(self, opt)
					current = opt
					self.CurrentOption = current
					pcall(dCallback, current)
				end
			}
			return ctrl
		end

		function tabObj:CreateInput(iData)
			local iName = "Input"
			local iDesc = ""
			local iPlaceholder = "Type here..."
			local iCallback = function() end
			if type(iData) == "table" then
				iName = tostring(iData.Name or iData.Title or "Input")
				iDesc = tostring(iData.Desc or iData.Description or "")
				iPlaceholder = tostring(iData.PlaceholderText or iData.Placeholder or iData.Default or "")
				iCallback = iData.Callback or function() end
			else
				iName = tostring(iData or "Input")
			end

			local current = ""
			if pTab and pTab.CreateInput then
				pTab:CreateInput(iName, iDesc, iPlaceholder, nil, nil, function(text)
					current = text
					pcall(iCallback, text)
				end)
			end

			return {
				Set = function(self, val)
					current = tostring(val)
					pcall(iCallback, current)
				end
			}
		end

		function tabObj:CreateKeybind(kData)
			local kName = "Keybind"
			local kDesc = ""
			local defaultKey = "None"
			local kCallback = function() end
			if type(kData) == "table" then
				kName = tostring(kData.Name or kData.Title or "Keybind")
				kDesc = tostring(kData.Desc or kData.Description or "")
				defaultKey = tostring(kData.CurrentKeybind or kData.Default or "None")
				kCallback = kData.Callback or function() end
			else
				kName = tostring(kData or "Keybind")
			end

			if pTab and pTab.CreateKeybind then
				pTab:CreateKeybind(kName, kDesc, defaultKey, function(key)
					pcall(kCallback, key)
				end)
			end

			return { Set = function() end }
		end

		function tabObj:CreateDivider()
			if pTab and pTab.CreateDivider then
				return pTab:CreateDivider()
			end
		end

		return tabObj
	end

	return v2
end

local function func24(cfgName, showNotify)
	cfgName = tostring(cfgName or "default"):gsub("[^%w_%-]", "")
	if cfgName == "" then cfgName = "default" end
	local filePath = "TPS_Configs/" .. cfgName .. ".json"
	local v4 = nil
	local readOk = pcall(function()
		if type(isfile) == "function" and isfile(filePath) then
			v4 = readfile(filePath)
		elseif type(readfile) == "function" then
			v4 = readfile(filePath)
		end
	end)
	if not readOk or not v4 or v4 == "" then
		if cfgName == "default" then
			pcall(function()
				if type(isfile) == "function" and isfile("TPS_Configs/Config.json") then
					v4 = readfile("TPS_Configs/Config.json")
				end
			end)
		end
	end
	if not v4 or v4 == "" then
		if showNotify then
			v1:Notify({Title = "Config File Not Found", Content = cfgName .. ".json does not exist", Type = "Warning", Duration = 2})
		end
		return false
	end
	local v2, v3 = pcall(function()
		return tbl8.decode(v4)
	end)
	if not v2 or type(v3) ~= "table" then
		if showNotify then
			v1:Notify({Title = "Config File Corrupted", Content = "Invalid JSON structure", Type = "Error", Duration = 2})
		end
		return false
	end
	func23(v3)
	if showNotify then
		v1:Notify({Title = "Config Loaded", Content = "Successfully loaded " .. cfgName, Type = "Success", Duration = 2})
	end
	return true
end

local function getSavedConfigsList()
	local list = {"default"}
	pcall(function()
		if type(isfolder) == "function" and not isfolder("TPS_Configs") then
			if type(makefolder) == "function" then makefolder("TPS_Configs") end
		end
		if type(listfiles) == "function" then
			local files = listfiles("TPS_Configs")
			for _, filePath in ipairs(files) do
				local name = filePath:match("([^\\/]+)%.json$")
				if name and name ~= "" and not table.find(list, name) then
					table.insert(list, name)
				end
			end
		end
	end)
	return list
end

local floatingIconGui = nil

local function toggleFloatingIcon(enable)
	if not enable then
		if floatingIconGui then
			floatingIconGui.Enabled = false
		end
		return
	end

	if floatingIconGui then
		floatingIconGui.Enabled = true
		return
	end

	local targetParent = nil
	pcall(function()
		if gethui then
			targetParent = gethui()
		elseif game:GetService("CoreGui") then
			targetParent = game:GetService("CoreGui")
		end
	end)
	if not targetParent then
		targetParent = LocalPlayer:FindFirstChildOfClass("PlayerGui") or LocalPlayer:WaitForChild("PlayerGui")
	end

	floatingIconGui = Instance.new("ScreenGui")
	floatingIconGui.Name = "TPS_MobileFloatingGui"
	floatingIconGui.ResetOnSpawn = false
	floatingIconGui.IgnoreGuiInset = true
	floatingIconGui.DisplayOrder = 99999
	floatingIconGui.Parent = targetParent

	local btn = Instance.new("TextButton")
	btn.Name = "FloatingButton"
	btn.Size = UDim2.new(0, 48, 0, 48)
	btn.Position = UDim2.new(0, 18, 0.5, -24)
	btn.BackgroundColor3 = Color3.fromRGB(20, 20, 26)
	btn.Text = "TPS"
	btn.Font = Enum.Font.GothamBold
	btn.TextSize = 14
	btn.TextColor3 = Color3.fromRGB(255, 255, 255)
	btn.AutoButtonColor = false
	btn.Active = true
	btn.Parent = floatingIconGui

	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(1, 0)
	corner.Parent = btn

	local stroke = Instance.new("UIStroke")
	stroke.Color = Color3.fromRGB(244, 63, 94)
	stroke.Thickness = 1.6
	stroke.Transparency = 0.25
	stroke.Parent = btn

	local dragging = false
	local dragStart = nil
	local startPos = nil
	local hasMoved = false

	btn.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			hasMoved = false
			dragStart = input.Position
			startPos = btn.Position

			local endConn
			endConn = input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
					if endConn then
						endConn:Disconnect()
					end
				end
			end)
		end
	end)

	btn.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = input.Position - dragStart
			if math.abs(delta.X) > 6 or math.abs(delta.Y) > 6 then
				hasMoved = true
			end
			btn.Position = UDim2.new(
				startPos.X.Scale,
				startPos.X.Offset + delta.X,
				startPos.Y.Scale,
				startPos.Y.Offset + delta.Y
			)
		end
	end)

	btn.Activated:Connect(function()
		if not hasMoved then
			if v2 and v2.Toggle then
				v2:Toggle()
			elseif win and win.Toggle then
				win:Toggle()
			end
		end
	end)

	pcall(function()
		getgenv()._TPSFloatingButton = btn
	end)
end
