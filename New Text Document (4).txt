    local function getGui()
        local function canParent(container)
            if not container then
                return false
            end
            local ok = pcall(
                function()
                    local probe = Instance.new("Folder")
                    probe.Name = "SpecterProbe"
                    probe.Parent = container
                    probe:Destroy()
                end
            )
            return ok
        end

        -- 1. Try gethui()
        local okHui, h = pcall(
            function()
                if gethui then
                    return gethui()
                end
            end
        )
        if okHui and h and canParent(h) then
            return h
        end

        -- 2. Try CoreGui (only if identity permits parenting, prevents Xeno level 3 security crash)
        local okCg, cg = pcall(
            function()
                return game:GetService("CoreGui")
            end
        )
        if okCg and cg and canParent(cg) then
            return cg
        end

        -- 3. Fallback to LocalPlayer PlayerGui
        local lp = LP or Players.LocalPlayer or game:GetService("Players").LocalPlayer
        if not lp then
            pcall(
                function()
                    lp = Players:GetPropertyChangedSignal("LocalPlayer"):Wait()
                end
            )
            lp = Players.LocalPlayer
        end
        if lp then
            local pGui = lp:FindFirstChild("PlayerGui") or lp:WaitForChild("PlayerGui", 5)
            if pGui and canParent(pGui) then
                return pGui
            end
        end

        -- 4. Ultimate fallbacks
        if okHui and h then
            return h
        end
        if lp and lp:FindFirstChild("PlayerGui") then
            return lp:FindFirstChild("PlayerGui")
        end
        if okCg and cg then
            return cg
        end
        return nil
    end

    -- Slider (Razor-Thin Futuristic Track + Micro Knob Expansion + Solid Black Badge)
    function TabMeta:AddSlider(id, opts)
        id, opts = _normOpts(id, opts)
        local T = self._W._T
        local minV, maxV = opts.Min or 0, opts.Max or 100
        local rnd = opts.Rounding
        local val = math.clamp(opts.Default or minV, minV, maxV)

        local _, ctrl = _row(self, opts.Title, opts.Description, 140)

        local valBadge = new(
            "Frame",
            {
                AnchorPoint = Vector2.new(1, 0.5), BackgroundColor3 = Color3.fromRGB(12, 13, 16),
                Position = UDim2.new(1, 0, 0.5, -9), Size = UDim2.fromOffset(38, 15), ZIndex = 16,
            }, ctrl
        )
        crn(4, valBadge)
        local vbStk = stk(Color3.fromRGB(48, 52, 60), 1, valBadge)
        vbStk.Transparency = 0.8

        local valLbl = new(
            "TextLabel",
            {
                BackgroundTransparency = 1, Size = UDim2.fromScale(1, 1),
                Font = Enum.Font.GothamBold, Text = tostring(val),
                TextColor3 = Color3.fromRGB(245, 247, 252), TextSize = 10, ZIndex = 17,
            }, valBadge
        )

        local track = new(
            "Frame",
            {
                AnchorPoint = Vector2.new(0, 0.5), BackgroundColor3 = Color3.fromRGB(20, 22, 28),
                Position = UDim2.new(0, 0, 0.5, 6), Size = UDim2.new(1, 0, 0, 4), ZIndex = 16,
            }, ctrl
        )
        crn(2, track)

        local frac0 = (maxV == minV) and 0 or math.clamp((val - minV) / (maxV - minV), 0, 1)

        local fill = new(
            "Frame",
            {BackgroundColor3 = T.Accent, Size = UDim2.new(frac0, 0, 1, 0), BorderSizePixel = 0, ZIndex = 17,},
            track
        )
        crn(2, fill)

        local kSz = 12
        local knob = new(
            "Frame",
            {
                AnchorPoint = Vector2.new(0.5, 0.5),
                BackgroundColor3 = Color3.fromRGB(255, 255, 255),
                Position = UDim2.new(frac0, 0, 0.5, 0), Size = UDim2.fromOffset(kSz, kSz),
                ZIndex = 19,
            }, track
        )
        crn(kSz / 2, knob)
        local kStk = stk(Color3.fromRGB(15, 15, 18), 1, knob)
        kStk.Transparency = 0.5

        local hit = new(
            "TextButton",
            {
                BackgroundTransparency = 1, Position = UDim2.new(0, -6, 0, -8),
                Size = UDim2.new(1, 12, 1, 16), Text = "", AutoButtonColor = false, ZIndex = 25,
            }, track
        )

        local obj = {Value = val, _id = id}
        local dragging = false

        local function applyPx(x)
            -- Guard: NaN input (some mobile executors pass nan for Position.X)
            if x ~= x then
                return
            end
            local tX = track.AbsolutePosition.X
            local tW = track.AbsoluteSize.X
            -- Guard: NaN or zero-size track (not yet laid out)
            if tX ~= tX or tW ~= tW or tW <= 1 then
                return
            end
            local rel = math.clamp((x - tX) / tW, 0, 1)
            local raw = minV + rel * (maxV - minV)
            -- Fix: in Lua 0 is truthy, so "rnd and ... / rnd" with rnd=0 causes raw/0 = nan
            -- Only use custom rounding when rnd is a positive number
            local final
            if rnd and rnd > 0 then
                final = math.floor(raw / rnd + 0.5) * rnd
            else
                final = math.floor(raw + 0.5)
            end
            if final ~= final then
                final = minV
            end -- fallback if still nan
            final = math.clamp(final, minV, maxV)
            obj.Value = final
            valLbl.Text = tostring(final)

            local fr = (maxV == minV) and 0 or ((final - minV) / (maxV - minV))
            fill.Size = UDim2.new(fr, 0, 1, 0)
            knob.Position = UDim2.new(fr, 0, 0.5, 0)

            if opts.Callback then
                pcall(opts.Callback, final)
            end
        end

        hit.InputBegan:Connect(
            function(i)
                if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
                    dragging = true
                    tw(knob, {Size = UDim2.fromOffset(14, 14)}, 0.1, Enum.EasingStyle.Quad)
                    applyPx(i.Position.X)
                end
            end
        )

        UserInputService.InputChanged:Connect(
            function(i)
                if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
                    applyPx(i.Position.X)
                end
            end
        )

        UserInputService.InputEnded:Connect(
            function(i)
                if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
                    dragging = false
                    tw(knob, {Size = UDim2.fromOffset(12, 12)}, 0.12, Enum.EasingStyle.Quad)
                end
            end
        )

        function obj:SetValue(v)
            v = math.clamp(v, minV, maxV)
            self.Value = v
            valLbl.Text = tostring(v)
            local fr = (maxV == minV) and 0 or ((v - minV) / (maxV - minV))
            tw(fill, {Size = UDim2.new(fr, 0, 1, 0)}, 0.14, Enum.EasingStyle.Quad)
            tw(knob, {Position = UDim2.new(fr, 0, 0.5, 0)}, 0.14, Enum.EasingStyle.Quad)
            if opts.Callback then
                pcall(opts.Callback, v)
            end
        end

        if id then
            self._W._flags[id] = obj
        end
        return obj
    end

    function Specter.SafeFireTouch(ball, part, phase)
        if not ball or not part then
            return false
        end
        if _firetouchinterest then
            return pcall(_firetouchinterest, ball, part, phase or 0)
        else
            -- Xeno CFrame Touch Simulation
            if phase == 0 or phase == nil then
                return pcall(function()
                    if part:IsA("BasePart") and ball:IsA("BasePart") then
                        local oldCF = part.CFrame
                        part.CFrame = ball.CFrame
                        task.delay(0.016, function()
                            if part and part.Parent then
                                part.CFrame = oldCF
                            end
                        end)
                    end
                end)
            end
            return true
        end
    end

    function Specter.SafeSetFFlag(name, val)
        if not _setfflag then
            return false
        end
        local function strip(k)
            return k:gsub("^DFInt", ""):gsub("^DFFlag", ""):gsub("FString", ""):gsub("FLog", ""):gsub(
                "^FFlag", ""
            ):gsub("^DFint", ""):gsub("^FInt", "")
        end
        return pcall(
            function()
                local ck = strip(name)
                if _getfflag and _getfflag(ck) ~= nil then
                    _setfflag(ck, tostring(val))
                elseif _getfflag and _getfflag(name) ~= nil then
                    _setfflag(name, tostring(val))
                end
            end
        )
    end

        if type(themeName) ~= "string" or type(themeTable) ~= "table" then
            return false

        THEMES[themeName] = themeTable
        return true
    end

    function Specter.CleanPrevious(names)
        names = names or {"SpecterUI", "Specter_MobileFAB", "Specter_MobileToggle"}
        local nameSet = {}
        for _, nm in ipairs(names) do
            nameSet[nm] = true
        end
        local containers = {}
        pcall(
            function()
                if gethui then
                    table.insert(containers, gethui())
                end
            end
        )
        pcall(
            function()
                table.insert(containers, game:GetService("CoreGui"))
            end
        )
        pcall(
            function()
                local lp = LP or Players.LocalPlayer
                if lp then
                    local pg = lp:FindFirstChildOfClass("PlayerGui") or lp:FindFirstChild(
                        "PlayerGui"
                    )
                    if pg then
                        table.insert(containers, pg)
                    end
                end
            end
        )
        for _, c in ipairs(containers) do
            if c then
                pcall(
                    function()
                        for _, child in ipairs(c:GetChildren()) do
                            if child:IsA("ScreenGui") and nameSet[child.Name] then
                                pcall(
                                    function()
                                        child:Destroy()
                                    end
                                )
                            end
                        end
                    end
                )
            end
        end
    end

        -- Tab Switching Physics
        function W:SelectTab(n)
            for i, t in ipairs(self._tabs) do
                local on = (i == n)
                local b = self._tabBtns[i]
                if b then
                    tw(
                        b,
                        {
                            BackgroundColor3 = on and Color3.fromRGB(18, 20, 26) or Color3.fromRGB(
                                10, 10, 14
                            ), BackgroundTransparency = on and 0.15 or 1,
                        }, 0.14, Enum.EasingStyle.Quad
                    )
                    local bStk = b:FindFirstChildWhichIsA("UIStroke")
                    if bStk then
                        tw(
                            bStk,
                            {
                                Color = on and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(
                                    40, 42, 48
                                ), Transparency = on and 0.88 or 1,
                            }, 0.14, Enum.EasingStyle.Quad
                        )
                    end
                    local lbl = b:FindFirstChild("Lbl")
                    if lbl then
                        tw(
                            lbl,
                            {
                                TextColor3 = on and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(
                                    175, 182, 195
                                ),
                                Position = on and UDim2.new(0, 14, 0, 0) or UDim2.new(0, 11, 0, 0),
                            }, 0.14, Enum.EasingStyle.Quad
                        )
                        lbl.Font = on and Enum.Font.GothamBold or Enum.Font.GothamMedium
                    end
                    local bar = b:FindFirstChild("Bar")
                    if bar then
                        tw(
                            bar,
                            {
                                BackgroundTransparency = on and 0 or 1,
                                Size = on and UDim2.new(0, 2, 0.55, 0) or UDim2.new(0, 2, 0.2, 0),
                            }, 0.14, Enum.EasingStyle.Quad
                        )
                    end
                end
                if t._tabContainer then
                    t._tabContainer.Visible = on
                    if on then
                        t._tabContainer.Position = UDim2.new(0, 0, 0, 4)
                        tw(
                            t._tabContainer, {Position = UDim2.new(0, 0, 0, 0)}, 0.14,
                            Enum.EasingStyle.Quad
                        )
                        if t._scroll then
                            _updateCanvas(t._scroll)
                        end
                        if t._activeSub and t._activeSub._scroll then
                            _updateCanvas(t._activeSub._scroll)
                        end
                    end
                end
            end
            self._active = self._tabs[n]
        end

            self._T = newT

        function W:SaveConfig(name)
            name = (name or "default"):gsub("[^%w_%-]", "")
            if name == "" then
                name = "default"

            local data = {}
            for flag, control in pairs(self._flags) do
                if control and control.Value ~= nil then

                        data[flag] = {

                            b = control.Value.B
                        }
                    elseif typeof(control.Value) == "EnumItem" then
                        data[flag] = {__type = "EnumItem", name = control.Value.Name}
                    else
                        data[flag] = control.Value
                    end
                end
            end

                    return HttpService:JSONEncode(data)
                end

                )
                return false
            end

            local path = configFolder .. "/" .. name .. ".json"
            local okSave = false

                function()
                    if writefile then
                        writefile(path, encoded)
                        okSave = true
                    end
                end
            )
            if okSave then
                self:Notify(
                    {Title = "Config Saved", Content = "Saved profile: " .. name, Kind = "ok"}
                )
                return true
            else
                self:Notify(
                    {Title = "Config Notice", Content = "writefile not supported on this executor.", Kind = "warn"}
                )
                return false
            end

        function W:LoadConfig(name)
            name = (name or "default"):gsub("[^%w_%-]", "")
            if name == "" then
                name = "default"
            end
            local path = configFolder .. "/" .. name .. ".json"
            local content
            local okRead = false
            pcall(
                function()
                    if isfile and readfile and isfile(path) then
                        content = readfile(path)
                        okRead = true
                    end
                end
            )
            if not okRead or not content then
                self:Notify(
                    {Title = "Config Notice", Content = "Config not found: " .. name, Kind = "warn"}
                )
                return false
            end
            local okJson, decoded = pcall(
                function()
                    return HttpService:JSONDecode(content)
                end
            )
            if not okJson or type(decoded) ~= "table" then
                self:Notify(
                    {Title = "Config Error", Content = "Failed to parse config file.", Kind = "err"}
                )
                return false
            end

            for flag, val in pairs(decoded) do
                local ctrl = self._flags[flag]
                if ctrl and ctrl.SetValue then
                    if type(val) == "table" and val.__type == "Color3" then
                        ctrl:SetValue(Color3.new(val.r, val.g, val.b))
                    elseif type(val) == "table" and val.__type == "EnumItem" and val.name then
                        local okEnum, kCode = pcall(
                            function()
                                return Enum.KeyCode[val.name]
                            end
                        )
                        if okEnum and kCode then
                            ctrl:SetValue(kCode)
                        end
                    else
                        ctrl:SetValue(val)
                    end
                end
            end
            self:Notify(
                {Title = "Config Loaded", Content = "Applied profile: " .. name, Kind = "ok"}
            )
            return true
        end

-- Smart Predictive Buffer
local BUF_MAX = 8
local POS_BUF = {}
for i = 1, BUF_MAX do
    POS_BUF[i] = {t = 0, pos = V3zero, vel = V3zero}
end
local POS_HEAD, POS_LEN = 0, 0

local function recordBall(ball)
    POS_HEAD = (POS_HEAD % BUF_MAX) + 1
    local s = POS_BUF[POS_HEAD]
    s.t = clock()
    s.pos = ball.Position
    s.vel = ball.AssemblyLinearVelocity
    if POS_LEN < BUF_MAX then
        POS_LEN = POS_LEN + 1
    end
end

local function getFrame(offset)
    return POS_BUF[((POS_HEAD - offset - 1) % BUF_MAX) + 1]
end

local function smartPredict(ball, ahead)
    if POS_LEN < 2 then
        if not ball then
            return V3zero, V3zero
        end
        return ball.Position + ball.AssemblyLinearVelocity * ahead, ball.AssemblyLinearVelocity
    end
    local s2 = getFrame(0)
    local s1 = getFrame(1)
    local dt = s2.t - s1.t
    if dt <= 0 then
        dt = 0.016
    end
    local vpx = (s2.pos.X - s1.pos.X) / dt
    local vpy = (s2.pos.Y - s1.pos.Y) / dt
    local vpz = (s2.pos.Z - s1.pos.Z) / dt
    local vrx = s2.vel.X
    local vry = s2.vel.Y
    local vrz = s2.vel.Z
    local ddx = vrx - vpx
    local ddy = vry - vpy
    local ddz = vrz - vpz
    local kb = clamp((ddx * ddx + ddy * ddy + ddz * ddz) / (35 * 35), 0, 1)
    local blx = vpx * (1 - kb) + vrx * kb
    local bly = vpy * (1 - kb) + vry * kb
    local blz = vpz * (1 - kb) + vrz * kb
    local ax, ay, az = 0, 0, 0
    if POS_LEN >= 3 then
        local s0 = getFrame(2)
        local dt0 = s1.t - s0.t
        if dt0 > 0 then
            local v0x = (s1.pos.X - s0.pos.X) / dt0
            local v0y = (s1.pos.Y - s0.pos.Y) / dt0
            local v0z = (s1.pos.Z - s0.pos.Z) / dt0
            local amax = 120
            ax = clamp((vpx - v0x) / dt, -amax, amax)
            ay = clamp((vpy - v0y) / dt, -amax, amax)
            az = clamp((vpz - v0z) / dt, -amax, amax)
        end
    end
    local gravY = -9.8 * ahead * ahead * (1 - kb)
    local t = clamp(ahead, 0.016, 0.25)
    local half = 0.5 * t * t
    return
        V3(
            s2.pos.X + blx * t + ax * half, s2.pos.Y + bly * t + ay * half + gravY,
            s2.pos.Z + blz * t + az * half
        ), V3(blx, bly, blz)
end

local _cachedPing = 0.05
local _lastPingT = 0
local HIGH_PING_MS = 0.10
local GRAV_HALF_SQ = -9.8 * 0.5 * (1 / 60) * (1 / 60)
local INSIDE_THRESH = 2.5

local function startReach2()
    if reachConn then
        reachConn:Disconnect()
    end
    reachPart.Size = reachSize
    reachConn = RunService.PreSimulation:Connect(
        function()
            if not _root or not _hum then
                return
            end
            local ball = getBall()
            if not ball then
                return
            end
            local now = clock()
            if now - _lastPingT >= 2 then
                _lastPingT = now
                pcall(
                    function()
                        _cachedPing = clamp(lp:GetNetworkPing(), 0.016, 0.25)
                    end
                )
            end
            local ballPos = ball.Position
            local rootPos = _root.Position
            local qdx = ballPos.X - rootPos.X
            local qdz = ballPos.Z - rootPos.Z
            if abs(qdx) > 16 or abs(qdz) > 16 then
                return
            end
            recordBall(ball)
            local qdy = ballPos.Y - rootPos.Y
            local rawDist = sqrt(qdx * qdx + qdy * qdy + qdz * qdz)
            local ahead = _cachedPing + 1 / 60
            local predPos, predVel
            if _cachedPing >= HIGH_PING_MS or rawDist >= 7 then
                predPos, predVel = smartPredict(ball, ahead)
            else
                predPos = ballPos
                predVel = ball.AssemblyLinearVelocity
            end
            local pvx = predVel.X
            local pvy = predVel.Y
            local pvz = predVel.Z
            local farPos = V3(
                predPos.X + pvx * (1 / 60), predPos.Y + pvy * (1 / 60) + GRAV_HALF_SQ,
                predPos.Z + pvz * (1 / 60)
            )
            local rx = reachSize.X / 2
            local ry = reachSize.Y / 2
            local rz = reachSize.Z / 2
            local dx = predPos.X - rootPos.X
            local dy = predPos.Y - rootPos.Y
            local dz = predPos.Z - rootPos.Z
            local fdx = farPos.X - rootPos.X
            local fdy = farPos.Y - rootPos.Y
            local fdz = farPos.Z - rootPos.Z
            if rawDist < INSIDE_THRESH then
                fireLeg(ball)
            end
            if (abs(dx) <= rx and abs(dy) <= ry and abs(dz) <= rz) or (abs(fdx) <= rx and abs(fdy) <= ry and abs(
                fdz
            ) <= rz) then
                fireLeg(ball)
            end
            reachPart.CFrame = _root.CFrame
        end
    )
end

local currentReactThread = nil

local function applyReact(tt, interp, name)
    if currentReactThread then
        task.cancel(currentReactThread)
        currentReactThread = nil
    end
    activeReactName = name or "Custom"
    currentReactThread = task.spawn(
        function()
            local flags = {
                ["DFIntTargetTimeDelayFacctorTenths"] = tt,
                ["FIntInterpolationMaxDelayMSec"] = interp,
            }
            local function apply()
                if not setfflag then
                    return
                end
                for k, v in pairs(flags) do
                    local ck = stripFlag(k)
                    pcall(
                        function()
                            if getfflag and getfflag(ck) ~= nil then
                                setfflag(ck, v)
                            elseif getfflag and getfflag(k) ~= nil then
                                setfflag(k, v)
                            end
                        end
                    )
                end
            end
            while true do
                apply()
                task.wait(20)
            end
        end
    )
    notify("React", "Activated " .. activeReactName .. " (TT: " .. tt .. ", Interp: " .. interp .. "ms)", 2, "ok")
end

local function isBoxTouching(pCF, pSz, tCF, tSz)
    local pMin = pCF.Position - pSz / 2
    local pMax = pCF.Position + pSz / 2
    local tMin = tCF.Position - tSz / 2
    local tMax = tCF.Position + tSz / 2
    return (pMin.X <= tMax.X and pMax.X >= tMin.X)
       and (pMin.Y <= tMax.Y and pMax.Y >= tMin.Y)
       and (pMin.Z <= tMax.Z and pMax.Z >= tMin.Z)
end

    end
    local function writeInterp(val)
        if not setfflag then return end
        pcall(function()
            local ck = stripFlag("FIntInterpolationMaxDelayMSec")
            if getfflag and getfflag(ck) ~= nil then
                setfflag(ck, val)
            elseif getfflag and getfflag("FIntInterpolationMaxDelayMSec") ~= nil then
                setfflag("FIntInterpolationMaxDelayMSec", val)
            end
        end)
    end

    headMossThread = task.spawn(function()
        while true do
            writeTT(tt_idle)
            writeInterp(interp)
            task.wait(20)
        end
    end)

    local _head, headContact = nil, false
    local function refreshChar(char)
        if not char then return end
        _head = char:FindFirstChild("Head")
    end
    refreshChar(lp.Character)
    lp.CharacterAdded:Connect(function(c)
        task.wait()
        refreshChar(c)
        c.ChildAdded:Connect(function() refreshChar(c) end)
    end)

    headMossConn = RunService.PreSimulation:Connect(function()
        if not _head then return end
        local ball = getBall()
        if not ball then return end
        local bp = ball.Position
        local hp = _head.Position
        local dx = bp.X - hp.X
        local dy = bp.Y - hp.Y
        local dz = bp.Z - hp.Z
        if dx >= -HX and dx <= HX and dz >= -HZ and dz <= HZ and dy >= 1.0 and dy <= HY then
            fti(ball, _head, 0)
            fti(ball, _head, 1)
            if not headContact then
                headContact = true
                writeTT(tt_hit)
            end
        else
            if headContact then
                headContact = false
                writeTT(tt_idle)
            end
        end
    end)
    notify("Head Moss", "Activated " .. (label or "Kersai Moss"), 2, "ok")
end

        headMossConn = nil
    end
    if headMossThread then
        task.cancel(headMossThread)
        headMossThread = nil
    end
    notify("Head Moss", "Stopped Kersai Moss", 2, "warn")
end

-- TAB 4: FREEWILL (Directional Auto-Kick)

local freewillLoop = nil
local FW_REACH = 5
local FW_HEIGHT = 3.1

local function getFreewillBall()
    local tps = workspace:FindFirstChild("TPSSystem")
    return (tps and tps:FindFirstChild("TPS")) or workspace:FindFirstChild("Ball")
end

local function directionalKick(hrp, ball)
    local lookDir = Camera.CFrame.LookVector
    hrp.CFrame = CFrame.new(hrp.Position, hrp.Position + Vector3.new(lookDir.X, 0, lookDir.Z))
    for i = 1, 25 do
        fti(hrp, ball, 0)
        fti(hrp, ball, 1)
    end
end

local function setFreewillState(state)
    if state then
        if freewillLoop then
            freewillLoop:Disconnect()
        end
        freewillLoop = RunService.Heartbeat:Connect(
            function()
                local char = lp.Character
                local hrp = char and char:FindFirstChild("HumanoidRootPart")
                local ball = getFreewillBall()
                if hrp and ball then
                    hrp.AssemblyLinearVelocity = Vector3.zero
                    local camLook = Camera.CFrame.LookVector
                    hrp.CFrame = CFrame.new(
                        ball.Position + Vector3.new(0, FW_HEIGHT, 0),
                        ball.Position + Vector3.new(camLook.X, FW_HEIGHT, camLook.Z)
                    )
                    if (hrp.Position - ball.Position).Magnitude <= FW_REACH then
                        directionalKick(hrp, ball)
                    end
                end
            end
        )
    else
        if freewillLoop then
            freewillLoop:Disconnect()
            freewillLoop = nil
        end
        if lp.Character and lp.Character:FindFirstChild("HumanoidRootPart") then
            lp.Character.HumanoidRootPart.Anchored = false
        end
    end
end

local function st_gk_get_limbs()
    local limb_names = {
        "LeftUpperLeg", "LeftLowerLeg", "LeftFoot", "Left Toes", "RightUpperLeg", "RightLowerLeg",
        "RightFoot", "Right Toes", "LeftUpperArm", "LeftLowerArm", "LeftHand", "RightUpperArm",
        "RightLowerArm", "RightHand", "HumanoidRootPart", "Head", "Torso", "UpperTorso",
        "LowerTorso"
    }
    local limbs = {}
    local char = LP.Character
    if not char then
        return limbs
    end
    for _, name in ipairs(limb_names) do
        local part = char:FindFirstChild(name)
        if part and part:IsA("BasePart") then
            table.insert(limbs, part)
        end
    end
    return limbs
end

local function st_gk_loop()
    st_gk_frame_counter = st_gk_frame_counter + 1
    if st_gk_frame_counter % 2 ~= 0 then
        return
    end
    if st_gk_anti_ban and math.random(1, 10) <= 3 then
        return
    end

    local ball = getBall()
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if ball and hrp then
        local dist = (hrp.Position - ball.Position).Magnitude
        local effGkDist = math.clamp(st_gk_dist, 1, 7)
        if dist <= effGkDist then
            local limbs = st_gk_get_limbs()
            for _, limb in ipairs(limbs) do
                fti(ball, limb, 0)
                fti(ball, limb, 1)
            end
        end
    end
end

local sr_enabled = false
local sr_thread = nil
local sr_distance = 3
local sr_last_activation = 0
local sr_running = false

local function sr_loop()
    if sr_running then
        return
    end
    sr_running = true
    while sr_enabled do
        local ball = getBall()
        local char = LP.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if ball and hrp then
            local dist = (hrp.Position - ball.Position).Magnitude
            local effSrDist = math.clamp(sr_distance, 1, 7)
            if dist <= effSrDist then
                local current_time = tick()
                if current_time - sr_last_activation > 0.2 then
                    local parts = {
                        char:FindFirstChild("HumanoidRootPart"), char:FindFirstChild("Head"),
                        char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
                    }
                    for _, part in ipairs(parts) do
                        if part then
                            fti(ball, part, 0);
                            fti(ball, part, 1)
                        end
                    end
                    sr_last_activation = current_time
                end
            end
        end
        task.wait(0.033)
    end
    sr_running = false
end
